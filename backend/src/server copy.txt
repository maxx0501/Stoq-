app.post('/auth/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    const user = await prisma.user.findUnique({ where: { email }, include: { memberships: true } });
    if (!user) return res.status(400).json({ error: "Usuário não encontrado." });
    const isPasswordValid = await bcrypt.compare(password, user.passwordHash);
    if (!isPasswordValid) return res.status(400).json({ error: "Senha incorreta." });
    const membership = user.memberships[0];
    const store = membership ? await prisma.store.findUnique({ where: { id: membership.storeId } }) : null;
    const token = jwt.sign(
      { userId: user.id, storeId: membership?.storeId, role: membership?.role, isSuperAdmin: user.isSuperAdmin, canSell: membership?.canSell, canManageProducts: membership?.canManageProducts }, 
      process.env.JWT_SECRET || 'segredo-padrao', { expiresIn: '7d' }
    );
    return res.json({ user: { name: user.name, email: user.email, avatarUrl: user.avatarUrl, isSuperAdmin: user.isSuperAdmin }, storeName: store?.name || 'Painel Admin', token });
  } catch (error) { return res.status(500).json({ error: "Erro interno no login." }); }
});

const getUserIdFromToken = (req: express.Request) => {
  const authHeader = req.headers.authorization;
  if (!authHeader) throw new Error('Sem token');
  const token = authHeader.split(' ')[1];
  return jwt.verify(token, process.env.JWT_SECRET || 'segredo-padrao') as any;
};

app.post('/sales', async (req, res) => {
  try {
    const user = getUserIdFromToken(req);
    if (user.role !== 'OWNER' && !user.canSell) return res.status(403).json({ error: "Sem permissão de venda." });
    
    // Recebe novos campos: customerId e paymentMethod
    const { items, customerId, paymentMethod } = req.body; 
    
    if (!items || items.length === 0) return res.status(400).json({ error: "Carrinho vazio." });

    // Lógica do Fiado
    let status = 'PAID';
    let dueDate = null;

    if (paymentMethod === 'CREDIT_STORE') {
        if (!customerId) return res.status(400).json({ error: "Para vender fiado, selecione um cliente!" });
        status = 'PENDING';
        
        // Define vencimento para 30 dias a partir de hoje
        const date = new Date();
        date.setDate(date.getDate() + 30);
        dueDate = date;
    }

    const sale = await prisma.$transaction(async (tx) => {
      let total = 0;
      
      // 1. Valida Estoque e Calcula Total
      for (const item of items) {
        const product = await tx.product.findUnique({ where: { id: item.productId } });
        if (!product || product.storeId !== user.storeId) throw new Error("Produto inválido.");
        if (product.stock < item.quantity) throw new Error(`Estoque insuficiente: ${product.name}`);
        total += Number(product.price) * item.quantity;
      }

      // 2. Cria a Venda
      const newSale = await tx.sale.create({
        data: {
          storeId: user.storeId,
          userId: user.userId, // Quem vendeu (se for admin, pega do token)
          customerId: customerId || null, // Vincula cliente
          total,
          paymentMethod: paymentMethod || 'MONEY',
          status: status as any,
          dueDate: dueDate
        }
      });

      // 3. Cria Itens e Baixa Estoque
      for (const item of items) {
        const product = await tx.product.findUnique({ where: { id: item.productId } });
        await tx.saleItem.create({ data: { saleId: newSale.id, productId: item.productId, quantity: item.quantity, price: product!.price } });
        await tx.product.update({ where: { id: item.productId }, data: { stock: { decrement: item.quantity } } });
      }
      
      return newSale;
    });

    return res.status(201).json(sale);
  } catch (error: any) { 
      return res.status(400).json({ error: error.message || "Erro na venda." }); 
  }
});

// --- ROTA EXTRA: BUSCAR DÍVIDAS DE UM CLIENTE ---
app.get('/customers/:id/debts', async (req, res) => {
    try {
        const user = getUserIdFromToken(req);
        const { id } = req.params;

        const debts = await prisma.sale.findMany({
            where: { 
                storeId: user.storeId, 
                customerId: id, 
                paymentMethod: 'CREDIT_STORE',
                status: 'PENDING'
            },
            orderBy: { createdAt: 'desc' }
        });

        // Calcula Juros (Simulação: 10% se atrasar)
        const debtsWithInterest = debts.map(sale => {
            const today = new Date();
            const due = new Date(sale.dueDate!);
            const isLate = today > due;
            
            // Juros simples de 10% fixo se estiver atrasado (pode refinar depois)
            const interest = isLate ? (Number(sale.total) * 0.10) : 0; 
            
            return {
                ...sale,
                isLate,
                interest,
                finalTotal: Number(sale.total) + interest
            };
        });

        return res.json(debtsWithInterest);
    } catch (error) { return res.status(500).json({ error: "Erro ao buscar dívidas." }); }
});

app.post('/cashflow/finish', async (req, res) => {
  try {
    const user = getUserIdFromToken(req);
    const { countedMoney } = req.body;
    const session = await prisma.cashSession.findFirst({ where: { storeId: user.storeId, OR: [{ status: 'OPEN' }, { status: 'CONFERENCE' }] }, orderBy: { startedAt: 'desc' }, include: { movements: true } });
    if (!session) return res.status(400).json({ error: 'Nenhuma sessão de caixa aberta.' });

    const movements = session.movements || [];
    const totalBleeds = movements.filter(m => m.type === 'BLEED').reduce((acc, m) => acc + Number(m.value), 0);
    const totalSupplies = movements.filter(m => m.type === 'SUPPLY').reduce((acc, m) => acc + Number(m.value), 0);

    // Soma vendas em dinheiro dentro do período da sessão
    const start = new Date(session.startedAt);
    const end = new Date();
    const salesSum = await prisma.sale.aggregate({
      _sum: { total: true },
      where: {
        storeId: user.storeId,
        paymentMethod: 'MONEY',
        createdAt: { gte: start, lte: end }
      }
    });
    const moneyTotal = Number(salesSum._sum.total || 0);

    const expected = Number(session.openingBalance) + moneyTotal + totalSupplies - totalBleeds;
    const diff = Number(countedMoney) - expected;

    const updated = await prisma.cashSession.update({ where: { id: session.id }, data: { countedMoney: Number(countedMoney), difference: diff, status: 'FINISHED', closedAt: new Date(), totalRevenue: moneyTotal, totalBleed: totalBleeds, totalSupply: totalSupplies } });
    return res.json({ ...updated, theoreticalBalance: expected });
  } catch (error) { console.error(error); return res.status(500).json({ error: 'Erro ao finalizar conferência.' }); }
});

    const formatted = await Promise.all(sessions.map(async (s) => {
      const movements = s.movements || [];
      const openMov = movements.find(m => m.type === 'OPEN');
      const operator = openMov?.desc || '-';
      const initialValue = Number(s.openingBalance ?? openMov?.value ?? 0);

      const start = new Date(s.startedAt);
      const end = s.closedAt ? new Date(s.closedAt) : now;
      const salesMoney = await prisma.sale.findMany({ where: { storeId: user.storeId, paymentMethod: 'MONEY', createdAt: { gte: start, lte: end } } });
      const moneyTotal = salesMoney.reduce((acc, sale) => acc + Number(sale.total), 0);

      const totalBleeds = movements.filter(m => m.type === 'BLEED').reduce((acc, m) => acc + Number(m.value), 0);
      const totalSupplies = movements.filter(m => m.type === 'SUPPLY').reduce((acc, m) => acc + Number(m.value), 0);

      const expectedValue = initialValue + moneyTotal + totalSupplies - totalBleeds;
      return {
        id: s.id,
        date: s.startedAt,
        operator,
        initial: initialValue,
        expected: expectedValue,
        counted: Number(s.countedMoney || 0),
        diff: Number(s.difference || 0),
        status: s.status
      };
    }));
    return res.json(formatted);
  } catch (error) { console.error(error); return res.status(500).json({ error: 'Erro ao buscar histórico' }); }
});

app.post('/stock/entry', async (req, res) => {
  try {
    const user = getUserIdFromToken(req);
    if (user.role !== 'OWNER' && !user.canManageProducts) return res.status(403).json({ error: "Sem permissão." });
    const { productId, quantity, newCostPrice, type, reason } = req.body;
    const qty = Number(quantity);
    if (!productId || qty <= 0) return res.status(400).json({ error: "Dados inválidos." });
    const currentProduct = await prisma.product.findUnique({ where: { id: productId } });
    if (!currentProduct) return res.status(404).json({ error: "Produto não encontrado." });
    const oldStock = currentProduct.stock;
    const isLoss = type === 'LOSS';
    const newStock = isLoss ? oldStock - qty : oldStock + qty;
    if (isLoss && newStock < 0) return res.status(400).json({ error: "Estoque insuficiente para essa baixa." });
    const result = await prisma.$transaction(async (tx) => {
      const updated = await tx.product.update({ where: { id: productId }, data: { stock: isLoss ? { decrement: qty } : { increment: qty }, costPrice: (!isLoss && newCostPrice) ? Number(newCostPrice) : undefined } });
      await tx.stockEntry.create({ data: { storeId: user.storeId, productId: productId, quantity: qty, oldStock: oldStock, newStock: newStock, costPrice: newCostPrice ? Number(newCostPrice) : currentProduct.costPrice, entryType: type || 'ENTRY', reason: reason || (isLoss ? 'Perda/Quebra' : 'Compra'), userId: user.id } });
      return updated;
    });
    return res.json(result);
  } catch (error) { return res.status(500).json({ error: "Erro na movimentação." }); }
});

    const user = getUserIdFromToken(req); const { period } = req.query; const today = new Date(); let startDate = new Date(); if (period === '30days') startDate.setDate(today.getDate() - 30); else if (period === 'year') startDate.setFullYear(today.getFullYear() - 1); else startDate.setDate(today.getDate() - 7);
    const sales = await prisma.sale.findMany({ where: { storeId: user.storeId, createdAt: { gte: startDate } }, include: { items: { include: { product: true } } }, orderBy: { createdAt: 'asc' } });
    let totalRevenue = 0; let totalCost = 0; const categoryMap: any = {};
    sales.forEach(sale => { totalRevenue += Number(sale.total); sale.items.forEach(item => { const cost = Number(item.product.costPrice || 0) * item.quantity; totalCost += cost; const cat = item.product.category || 'Outros'; if (!categoryMap[cat]) categoryMap[cat] = 0; categoryMap[cat] += Number(item.price) * item.quantity; }); });
    const netProfit = totalRevenue - totalCost; const margin = totalRevenue > 0 ? ((netProfit / totalRevenue) * 100).toFixed(1) : 0;
    const salesByDay: any = {}; const profitByDay: any = {};
    sales.forEach(sale => { const dateKey = new Date(sale.createdAt).toLocaleDateString('pt-BR'); if (!salesByDay[dateKey]) { salesByDay[dateKey] = 0; profitByDay[dateKey] = 0; } salesByDay[dateKey] += Number(sale.total); let saleCost = 0; sale.items.forEach(i => saleCost += Number(i.product.costPrice || 0) * i.quantity); profitByDay[dateKey] += (Number(sale.total) - saleCost); });
    const chartData = Object.keys(salesByDay).map(date => ({ date, sales: salesByDay[date], profit: profitByDay[date] }));
    const categoryData = Object.keys(categoryMap).map(name => ({ name, value: categoryMap[name] })).sort((a, b) => b.value - a.value);
    return res.json({ totalRevenue, netProfit, margin, chartData, categoryData });
  } catch (error) { return res.status(500).json({ error: "Erro ao gerar relatório." }); }
});
